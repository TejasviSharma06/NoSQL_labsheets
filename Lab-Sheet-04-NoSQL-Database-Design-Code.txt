PRACTICAL 8 — HBase sessions with TTL
============================================================
[HBase Shell]

# TTL is configured in seconds. This example uses 3600 seconds (1 hour).
create 'UserSessions', {NAME => 'session', TTL => 3600}, 'user'

put 'UserSessions', 'SESS001', 'session:token', 'token-abc'
put 'UserSessions', 'SESS001', 'session:created_at', '2026-10-13T09:00:00'
put 'UserSessions', 'SESS001', 'user:user_id', 'U101'

put 'UserSessions', 'SESS002', 'session:token', 'token-def'
put 'UserSessions', 'SESS002', 'session:created_at', '2026-10-13T09:05:00'
put 'UserSessions', 'SESS002', 'user:user_id', 'U102'

scan 'UserSessions'
# Expired cells are removed during HBase maintenance/compaction; expiration may not be immediate.
# To change TTL later (HBase versions that support this syntax):
alter 'UserSessions', {NAME => 'session', TTL => 3600}

PRACTICAL 9 — Row-key design for large-scale application
============================================================
[HBase Shell]

create 'LargeScaleEvents', 'event', 'user'

# Prefix/salted row keys can spread writes when there are many records.
put 'LargeScaleEvents', '00_EVT001', 'event:type', 'login'
put 'LargeScaleEvents', '01_EVT002', 'event:type', 'upload'
put 'LargeScaleEvents', '02_EVT003', 'event:type', 'logout'
put 'LargeScaleEvents', '03_EVT004', 'event:type', 'login'
put 'LargeScaleEvents', '00_EVT005', 'event:type', 'purchase'

put 'LargeScaleEvents', '00_EVT001', 'user:id', 'U101'
put 'LargeScaleEvents', '01_EVT002', 'user:id', 'U102'
put 'LargeScaleEvents', '02_EVT003', 'user:id', 'U103'
put 'LargeScaleEvents', '03_EVT004', 'user:id', 'U104'
put 'LargeScaleEvents', '00_EVT005', 'user:id', 'U101'

scan 'LargeScaleEvents'
# A well-designed row key can avoid write hotspots; pre-splitting and actual region
# layout must be configured/checked in the HBase cluster.

PRACTICAL 14 — Cassandra consistency levels
============================================================
[Run in cqlsh on a multi-node cluster; RF must support requested level.]

CREATE KEYSPACE IF NOT EXISTS ConsistencyDB
WITH replication = {'class': 'SimpleStrategy', 'replication_factor': 3};

USE ConsistencyDB;

CREATE TABLE IF NOT EXISTS ConsistencyDemo (
    record_id int PRIMARY KEY,
    note text
);

CONSISTENCY ONE;
INSERT INTO ConsistencyDemo (record_id, note) VALUES (1, 'write at ONE');
SELECT * FROM ConsistencyDemo WHERE record_id = 1;

CONSISTENCY QUORUM;
INSERT INTO ConsistencyDemo (record_id, note) VALUES (2, 'write at QUORUM');
SELECT * FROM ConsistencyDemo WHERE record_id = 2;

CONSISTENCY ALL;
INSERT INTO ConsistencyDemo (record_id, note) VALUES (3, 'write at ALL');
SELECT * FROM ConsistencyDemo WHERE record_id = 3;

# ONE needs one replica response; QUORUM needs a majority; ALL needs all replicas.
# With RF=3, QUORUM normally requires 2 replicas and ALL requires 3.
# On a single-node installation with RF=3, ALL/QUORUM operations may fail.

PRACTICAL 15 — Column-oriented application (HBase Event Logging)
============================================================
[HBase Shell]

create 'AppEventLog', 'event', 'user', 'status'

put 'AppEventLog', 'EVT001', 'event:timestamp', '2026-10-01T09:00:00'
put 'AppEventLog', 'EVT001', 'event:type', 'login'
put 'AppEventLog', 'EVT001', 'user:id', 'U101'
put 'AppEventLog', 'EVT001', 'status:result', 'success'

put 'AppEventLog', 'EVT002', 'event:timestamp', '2026-10-01T09:10:00'
put 'AppEventLog', 'EVT002', 'event:type', 'file-upload'
put 'AppEventLog', 'EVT002', 'user:id', 'U102'
put 'AppEventLog', 'EVT002', 'status:result', 'success'

get 'AppEventLog', 'EVT001'
scan 'AppEventLog'

# Update one cell
put 'AppEventLog', 'EVT001', 'status:result', 'verified'
get 'AppEventLog', 'EVT001'

# Delete one cell
delete 'AppEventLog', 'EVT002', 'status:result'
get 'AppEventLog', 'EVT002'

# HBase offers row-level atomic operations, sparse columns, and horizontal scaling
# through regions/RegionServers. It does not provide general multi-row SQL-style
# transactions. Querying is mainly by row key and scans/filters.
# Cassandra uses partition keys and replication; consistency is tunable per operation.
