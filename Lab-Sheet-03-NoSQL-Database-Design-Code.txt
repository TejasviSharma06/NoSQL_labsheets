  items: [

  ]
});

// Add a product

// Change quantity
db.Carts.updateOne({ userId: "u01", "items.productId": 101 }, { $set: { "items.$.quantity": 3 } });
// Remove a product
db.Carts.updateOne({ userId: "u01" }, { $pull: { items: { productId: 102 } } });
// Calculate cart value
db.Carts.aggregate([
  { $match: { userId: "u01" } }, { $unwind: "$items" },
  { $group: { _id: "$userId", cartValue: { $sum: { $multiply: ["$items.price", "$items.quantity"] } } } }
]);

// PRACTICAL 11: Bank Wallet Transfer using MongoDB transaction
// Requires a replica set/sharded cluster, not standalone MongoDB.
// ------------------------------------------------------------
db.Wallets.insertMany([
  { userId: "userA", balance: 5000 },
  { userId: "userB", balance: 2500 }
]);

// Run in mongosh. Transaction example:
const session = db.getMongo().startSession();
const txdb = session.getDatabase("NoSQLLabSheet3");
try {
  session.startTransaction();
  const debit = txdb.Wallets.updateOne(
    { userId: "userA", balance: { $gte: 500 } },
    { $inc: { balance: -500 } }
  );
  if (debit.modifiedCount !== 1) throw new Error("Insufficient balance or sender missing");
  const credit = txdb.Wallets.updateOne(
    { userId: "userB" },
    { $inc: { balance: 500 } }
  );
  if (credit.modifiedCount !== 1) throw new Error("Receiver missing");
  session.commitTransaction();
  print("Transfer successful");
} catch (e) {
  session.abortTransaction();
  print("Transfer failed: " + e.message);
} finally {
  session.endSession();
}
db.Wallets.find();
