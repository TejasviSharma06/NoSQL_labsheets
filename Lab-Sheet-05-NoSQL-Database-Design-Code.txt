PRACTICAL 10 — Riak set operations (application-side Python example)
============================================================
# Save as riak_set_operations.py. Install requests with: pip install requests
import requests

BASE = "http://localhost:8098/buckets/user_sets/keys"

def put_set(key, values):
    response = requests.put(
        f"{BASE}/{key}",
        json=sorted(set(values)),
        headers={"Content-Type": "application/json"}
    )
    response.raise_for_status()

def get_set(key):
    response = requests.get(f"{BASE}/{key}")
    response.raise_for_status()
    return set(response.json())

put_set("U101", ["music", "travel", "coding"])
put_set("U102", ["music", "reading", "coding"])

set1 = get_set("U101")
set2 = get_set("U102")

# Add an element
set1.add("sports")
put_set("U101", set1)

# Remove an element
set1.discard("travel")
put_set("U101", set1)

# Check membership and find common elements
print("coding in U101:", "coding" in get_set("U101"))
print("Common interests:", get_set("U101") & get_set("U102"))
print("U101 interests:", get_set("U101"))

  (cat:Category {name: 'Computer Accessories'}),
  (o1:Order {orderId: 1001, date: '2026-10-01'}),
  (o2:Order {orderId: 1002, date: '2026-10-02'}),
  (c1)-[:PLACED]->(o1),
  (c2)-[:PLACED]->(o2),
  (o1)-[:CONTAINS {quantity: 2}]->(p1),
  (o1)-[:CONTAINS {quantity: 1}]->(p2),
  (o2)-[:CONTAINS {quantity: 1}]->(p2),
  (p1)-[:IN_CATEGORY]->(cat),
  (p2)-[:IN_CATEGORY]->(cat);

# Products purchased by Aarav
MATCH (:Customer {name: 'Aarav'})-[:PLACED]->(:Order)-[:CONTAINS]->(p:Product)
RETURN DISTINCT p.name, p.price;

# Customers who purchased Keyboard
MATCH (c:Customer)-[:PLACED]->(:Order)-[:CONTAINS]->(:Product {name: 'Keyboard'})
RETURN DISTINCT c.name;

PRACTICAL 15 — Neo4j CRUD application using Python

# Save this section as neo4j_crud.py.
# Install: pip install neo4j
# Set NEO4J_URI, NEO4J_USERNAME, and NEO4J_PASSWORD for your own database.

import os
from neo4j import GraphDatabase

URI = os.getenv("NEO4J_URI", "bolt://localhost:7687")
USERNAME = os.getenv("NEO4J_USERNAME", "neo4j")
PASSWORD = os.getenv("NEO4J_PASSWORD", "change_this_password")

driver = GraphDatabase.driver(URI, auth=(USERNAME, PASSWORD))

def create_student(tx, student_id, name, course):
    tx.run(
        """
        MERGE (s:Student {studentId: $student_id})
        SET s.name = $name, s.course = $course
        """,
        student_id=student_id, name=name, course=course
    )

def get_students(tx):
    result = tx.run(
        "MATCH (s:Student) RETURN s.studentId AS id, s.name AS name, s.course AS course ORDER BY id"
    )
    return [record.data() for record in result]

def update_student(tx, student_id, name):
    tx.run(
        "MATCH (s:Student {studentId: $student_id}) SET s.name = $name",
        student_id=student_id, name=name
    )

def delete_student(tx, student_id):
    tx.run(
        "MATCH (s:Student {studentId: $student_id}) DETACH DELETE s",
        student_id=student_id
    )

try:
    with driver.session() as session:
        session.execute_write(create_student, 1, "Aarav", "BCA")
        session.execute_write(create_student, 2, "Priya", "BCA")
        print("Students:", session.execute_read(get_students))
        session.execute_write(update_student, 1, "Aarav Sharma")
        print("After update:", session.execute_read(get_students))
        # Uncomment to delete student 2:
        # session.execute_write(delete_student, 2)
finally:
    driver.close()

# Graph databases make relationship traversal natural using nodes and edges.
# Neo4j transactions provide ACID guarantees for committed graph updates.
# Scaling and availability depend on the selected Neo4j edition and cluster setup.
